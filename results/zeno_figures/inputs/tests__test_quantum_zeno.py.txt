"""Physical and independent numerical checks for the Z1 extension."""

import numpy as np
import pytest
from scipy.linalg import expm

from quantum_project import KET_00, KET_RR, PSI_PLUS, evolve_static, normalise_state, rydberg_hamiltonian
from quantum_project.density_matrix import (
    angular_frequency_variance, density_from_state, density_probability,
    evolve_density, projective_measurement, validate_density_matrix,
    validate_projectors,
)
from quantum_project.quantum_zeno import measurement_projectors, repeated_measurements
from quantum_project.units import mhz_to_angular_frequency

OMEGA = mhz_to_angular_frequency(1)
T = 0.5e-6
RHO0 = density_from_state(KET_00)


def hamiltonian(ratio=0, phase=0, delta=0):
    return rydberg_hamiltonian(OMEGA, delta * OMEGA, phase, ratio * OMEGA)


@pytest.mark.parametrize("ratio", [0, 1, 10, 100, -100])
@pytest.mark.parametrize("initial", [KET_00, PSI_PLUS, normalise_state([1, 2j, -3, 4j])])
def test_unmeasured_matches_milestone(ratio, initial):
    times = np.array([T, 0, -T, T / 3, T])
    hbar = hamiltonian(ratio, phase=0.73, delta=-0.4)
    pure = evolve_static(initial, times, hbar)
    expected = np.einsum("ti,tj->tij", pure, pure.conj())
    actual = evolve_density(density_from_state(initial), times, hbar)
    np.testing.assert_allclose(actual, expected, atol=2e-13, rtol=0)


@pytest.mark.parametrize("bad", [np.eye(3), np.eye(4), np.diag([1.1, -0.1, 0, 0]),
                                     np.full((4, 4), np.nan),
                                     RHO0 + 0.1j * np.eye(4),
                                     RHO0 + np.diag([0.1, -0.1, 0], 1)])
def test_invalid_density(bad):
    with pytest.raises(ValueError):
        validate_density_matrix(bad)


def test_density_tolerance_and_no_repair():
    rho = np.diag([1 + 1e-12, -1e-12, 0, 0]).astype(complex)
    np.testing.assert_array_equal(validate_density_matrix(rho), rho)
    with pytest.raises(ValueError):
        validate_density_matrix(rho, atol=1e-13)


@pytest.mark.parametrize("question", ["ground", "any_excitation", "double"])
@pytest.mark.parametrize("ratio", [0, 1, 100, -100])
def test_physical_density_invariants(question, ratio):
    mixed = 0.6 * RHO0 + 0.4 * density_from_state(PSI_PLUS)
    trajectory = repeated_measurements(mixed, hamiltonian(ratio), T, 37,
                                      np.linspace(0, T, 151), measurement_projectors(question))
    np.testing.assert_allclose(np.trace(trajectory, axis1=1, axis2=2), 1, atol=2e-12, rtol=0)
    np.testing.assert_allclose(trajectory, trajectory.conj().transpose(0, 2, 1), atol=2e-12, rtol=0)
    assert np.linalg.eigvalsh(trajectory).min() > -2e-12
    probabilities = trajectory.diagonal(axis1=1, axis2=2).real
    assert probabilities.min() >= -2e-12 and probabilities.max() <= 1 + 2e-12


def test_measurement_is_luders_channel_and_preserves_within_sector_coherence():
    state = normalise_state([1, 1j, 2, 3j])
    rho = density_from_state(state)
    actual = projective_measurement(rho, measurement_projectors())
    np.testing.assert_allclose(actual[1:, 1:], rho[1:, 1:], atol=1e-15)
    np.testing.assert_allclose(actual[0, 1:], 0, atol=1e-15)
    np.testing.assert_allclose(np.diag(actual), np.diag(rho), atol=1e-15)
    assert np.trace(actual @ actual).real < 1
    assert abs(actual[1, 2]) > 0.1
    np.testing.assert_allclose(projective_measurement(actual, measurement_projectors()), actual)


def test_a_and_b_are_identical_even_for_a_general_mixed_state():
    rho = 0.3 * np.eye(4) / 4 + 0.7 * density_from_state(normalise_state([1j, 2, 3j, 4]))
    a = repeated_measurements(rho, hamiltonian(3), T, 20, projectors=measurement_projectors("ground"))
    b = repeated_measurements(rho, hamiltonian(3), T, 20, projectors=measurement_projectors("any_excitation"))
    np.testing.assert_allclose(a, b, atol=1e-15, rtol=0)

def test_endpoint_convention_and_independent_channel_power():
    hbar = hamiltonian(1)
    for count in (0, 1, 2, 5, 20):
        actual = repeated_measurements(RHO0, hbar, T, count)[0]
        if count == 0:
            expected = evolve_density(RHO0, [T], hbar)[0]
        else:
            unitary = expm(-1j * hbar * T / count)
            # Column-major vectorisation: vec(K rho K†)=(K* tensor K)vec(rho).
            kraus = [p @ unitary for p in measurement_projectors()]
            channel = sum(np.kron(k.conj(), k) for k in kraus)
            expected = (np.linalg.matrix_power(channel, count) @ RHO0.ravel(order="F")).reshape(4, 4, order="F")
        np.testing.assert_allclose(actual, expected, atol=1e-13, rtol=0)
    unmeasured = repeated_measurements(RHO0, hbar, T, 0)[0]
    once = repeated_measurements(RHO0, hbar, T, 1)[0]
    assert density_probability(once, KET_00) == pytest.approx(density_probability(unmeasured, KET_00))


def test_samples_are_independent_of_grid_and_events_are_post_measurement():
    events = np.linspace(0, T, 6)
    sampled = repeated_measurements(RHO0, hamiltonian(1), T, 5, events)
    np.testing.assert_allclose(sampled[1:, 0, 1:], 0, atol=1e-15)
    dense = repeated_measurements(RHO0, hamiltonian(1), T, 5, np.linspace(0, T, 1001))
    np.testing.assert_array_equal(dense[-1], sampled[-1])
    unsorted = repeated_measurements(RHO0, hamiltonian(1), T, 5, events[::-1])
    np.testing.assert_array_equal(unsorted, sampled[::-1])


@pytest.mark.parametrize("ratio", [0, 1, 100, -100])
def test_actual_variance_and_quadratic_short_time_agreement(ratio):
    hbar = hamiltonian(ratio, phase=0.4, delta=0.3)
    variance = angular_frequency_variance(KET_00, hbar)
    assert variance == pytest.approx(OMEGA**2 / 2, rel=2e-15)
    times = np.array([0.2e-9, 0.1e-9])
    survival = evolve_density(RHO0, times, hbar)[:, 0, 0].real
    residual = survival - (1 - variance * times**2)
    assert np.max(abs(residual) / (variance * times**2)) < 1e-4
    assert 12 < residual[0] / residual[1] < 20  # leading t^4 remainder


@pytest.mark.parametrize("ratio", [0, 1, 100, -100])
def test_nonselective_zeno_limit_and_asymptotic_residual(ratio):
    counts = np.array([1000, 2000, 4000])
    survival = np.array([repeated_measurements(RHO0, hamiltonian(ratio), T, int(m))[0, 0, 0].real for m in counts])
    prediction = np.exp(-OMEGA**2 / 2 * T**2 / counts)
    assert np.all(np.diff(survival) > 0)
    assert survival[-1] > 0.998
    np.testing.assert_allclose((1 - survival[-1]) * counts[-1], OMEGA**2 / 2 * T**2, rtol=0.003)
    assert abs(survival[-1] - prediction[-1]) < abs(survival[0] - prediction[0]) / 8


def test_rank_two_nonselective_exact_solution_and_return_paths():
    hbar = hamiltonian()
    hbar[3, :] = 0
    hbar[:, 3] = 0  # Independent ideal-blockade test Hamiltonian; production stays intact.
    coupling = OMEGA / np.sqrt(2)
    for count in (2, 5, 100):
        result = repeated_measurements(RHO0, hbar, T, count)[0, 0, 0].real
        expected = (1 + np.cos(2 * coupling * T / count)**count) / 2
        all_yes = np.cos(coupling * T / count)**(2 * count)
        assert result == pytest.approx(expected, abs=1e-13)
        assert result > all_yes


def test_double_excitation_measurement_allows_zeno_subspace_dynamics():
    result = repeated_measurements(RHO0, hamiltonian(), T, 4000,
                                  projectors=measurement_projectors("double"))[0]
    assert density_probability(result, KET_RR) < 0.002
    assert density_probability(result, KET_00) == pytest.approx(np.cos(OMEGA * T / np.sqrt(2))**2, abs=0.002)
    assert density_probability(result, PSI_PLUS) > 0.6

def test_empty_zero_time_and_input_preservation():
    rho, hbar = RHO0.copy(), hamiltonian(1)
    original = hbar.copy()
    assert repeated_measurements(rho, hbar, T, 2, []).shape == (0, 4, 4)
    assert evolve_density(rho, [], hbar).shape == (0, 4, 4)
    np.testing.assert_allclose(repeated_measurements(rho, hbar, 0, 10)[0], rho)
    np.testing.assert_array_equal(hbar, original)
    np.testing.assert_array_equal(rho, RHO0)


@pytest.mark.parametrize("count", [10, 100])
def test_equivalent_display_grid_events_are_post_measurement(count):
    # The two linspace calls round some mathematically identical times differently.
    display = np.linspace(0, T, 1001)
    events = np.linspace(0, T, count + 1)
    event_indices = np.arange(count + 1) * (1000 // count)
    hbar = hamiltonian(1)
    sampled = repeated_measurements(RHO0, hbar, T, count, display[event_indices])
    expected = repeated_measurements(RHO0, hbar, T, count, events)
    np.testing.assert_array_equal(sampled, expected)
    np.testing.assert_array_equal(sampled[:, 0, 1:], 0)
    # A physically distinct pre-event sample must retain its coherence.
    offset = T / count * 1e-6
    before, at, after = repeated_measurements(
        RHO0, hbar, T, count, [events[1]-offset, events[1], events[1]+offset])
    assert np.max(abs(before[0, 1:])) > 1e-3
    np.testing.assert_array_equal(at[0, 1:], 0)
    assert np.max(abs(after[0, 1:])) > 1e-10


@pytest.mark.parametrize("mixed", [False, True])
def test_density_outer_product_and_unitary_purity(mixed):
    psi = normalise_state([1, 2j, -3, 4j])
    rho = density_from_state(psi)
    np.testing.assert_array_equal(rho, psi[:, None] * psi.conj()[None, :])
    np.testing.assert_allclose(rho @ rho, rho, atol=2e-16, rtol=0)
    if mixed:
        rho = .4 * rho + .6 * np.eye(4) / 4
    initial_purity = np.trace(rho @ rho).real
    output = evolve_density(rho, [0, T/7, T, -T], hamiltonian(100, .37, -.2))
    np.testing.assert_allclose(np.einsum('tij,tji->t', output, output), initial_purity, atol=2e-14, rtol=0)
    np.testing.assert_allclose(np.trace(output, axis1=1, axis2=2), 1, atol=2e-14, rtol=0)
    np.testing.assert_allclose(output, output.conj().transpose(0, 2, 1), atol=2e-14, rtol=0)
    assert np.linalg.eigvalsh(output).min() > -2e-14


@pytest.mark.parametrize("question,diagonal", [
    ("ground", [1, 0, 0, 0]), ("any_excitation", [0, 1, 1, 1]),
    ("double", [0, 0, 0, 1]),
])
def test_explicit_projectors_and_full_measurement_matrix(question, diagonal):
    p, q = measurement_projectors(question)
    np.testing.assert_array_equal(p, np.diag(diagonal))
    np.testing.assert_array_equal(q, np.eye(4)-p)
    for projector in (p, q):
        np.testing.assert_array_equal(projector @ projector, projector)
        np.testing.assert_array_equal(projector.conj().T, projector)
    np.testing.assert_array_equal(p @ q, np.zeros((4,4)))
    rho = density_from_state(normalise_state([1, 1j, 2, 3j]))
    same_block = np.equal.outer(diagonal, diagonal)
    expected = rho * same_block  # Explicit independent block matrix.
    actual = projective_measurement(rho, [p, q])
    np.testing.assert_array_equal(actual, expected)
    np.testing.assert_allclose(np.trace(actual), 1, atol=3e-16, rtol=0)
    np.testing.assert_array_equal(actual.conj().T, actual)


def test_selective_branch_probabilities_sum_to_nonselective_density():
    count = 3
    unitary = expm(-1j * hamiltonian(1, .3, .2) * T/count)
    branches = [KET_00.copy()]
    p, q = measurement_projectors()
    for _ in range(count):
        branches = [projector @ unitary @ psi for psi in branches for projector in (p, q)]
    # Unnormalised branch norm is its joint outcome probability; no reset assumed.
    total = sum(np.outer(psi, psi.conj()) for psi in branches)
    actual = repeated_measurements(RHO0, hamiltonian(1, .3, .2), T, count)[0]
    np.testing.assert_allclose(total, actual, atol=2e-15, rtol=0)
    all_yes = np.vdot(branches[0], branches[0]).real
    assert all_yes == pytest.approx(abs(unitary[0,0])**(2*count), abs=1e-15)
    assert actual[0,0].real > all_yes + .01


def test_short_time_prediction_does_not_call_propagation(monkeypatch):
    import quantum_project.density_matrix as dm
    def forbidden(*args, **kwargs):
        raise AssertionError("Variance must not depend on a propagator")
    monkeypatch.setattr(dm, "unitary_operator", forbidden)
    monkeypatch.setattr(dm, "expm", forbidden)
    hbar = hamiltonian(100, .3, -.4)
    direct = (hbar @ hbar)[0,0].real - hbar[0,0].real**2
    assert dm.angular_frequency_variance(KET_00, hbar) == pytest.approx(direct, rel=2e-15)


def test_saved_zeno_observables_reconstruct_from_complex_densities():
    from pathlib import Path
    path = Path(__file__).resolve().parents[1] / 'results/data/zeno_study.npz'
    with np.load(path, allow_pickle=False) as data:
        for prefix in ('baseline', 'trajectory', 'final'):
            rho = data[prefix+'_density']
            expected = np.stack([rho[...,0,0].real,
                (rho[...,1,1]+rho[...,2,2]+rho[...,1,2]+rho[...,2,1]).real/2,
                rho[...,3,3].real], axis=-1)
            np.testing.assert_allclose(data[prefix+'_probabilities'], expected, atol=1e-14, rtol=0)
            np.testing.assert_allclose(np.trace(rho,axis1=-2,axis2=-1),1,atol=2e-12,rtol=0)
        np.testing.assert_allclose(np.einsum('...ij,...ji->...',data['baseline_density'],data['baseline_density']),
                                   1,atol=2e-12,rtol=0)
